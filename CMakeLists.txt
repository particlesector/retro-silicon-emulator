if(RSE_BUILD_POLICY)
  find_path(RSE_ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${RSE_ONNXRUNTIME_ROOT}/include" REQUIRED)
  find_library(RSE_ORT_LIBRARY onnxruntime HINTS "${RSE_ONNXRUNTIME_ROOT}/lib" REQUIRED)
  add_library(rse_policy STATIC src/session/policy_playback.cpp)
  target_include_directories(rse_policy PRIVATE "${RSE_ORT_INCLUDE}")
  target_link_libraries(rse_policy PUBLIC rse_nes_session PRIVATE "${RSE_ORT_LIBRARY}")
  if(WIN32)
    find_file(RSE_ORT_DLL onnxruntime.dll HINTS "${RSE_ONNXRUNTIME_ROOT}/lib" REQUIRED)
  endif()
  function(rse_link_policy target)
    target_link_libraries(${target} PRIVATE rse_policy)
    target_compile_definitions(${target} PRIVATE RSE_POLICY_ENABLED=1)
    if(WIN32)
      add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E
        copy_if_different "${RSE_ORT_DLL}" "$<TARGET_FILE_DIR:${target}>")
    endif()
  endfunction()
  add_executable(rse-policy apps/cli/policy_main.cpp)
  rse_link_policy(rse-policy)
  install(TARGETS rse-policy RUNTIME DESTINATION .)
endif()
