cmake_minimum_required(VERSION 3.25)
project(RetroSiliconEmulator VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RSE_BUILD_TESTS "Build unit tests" ON)
option(RSE_BUILD_DESKTOP "Build SDL3 desktop frontend" OFF)
add_library(rse_core STATIC src/core/machine.cpp)
target_include_directories(rse_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(rse_core PUBLIC cxx_std_23)
add_library(rse_nes STATIC src/systems/nes/cartridge.cpp src/systems/nes/bus.cpp)
target_link_libraries(rse_nes PUBLIC rse_core)
add_executable(rse-cli apps/cli/main.cpp)
target_link_libraries(rse-cli PRIVATE rse_core)
if(RSE_BUILD_TESTS)
  enable_testing()
  add_executable(rse-core-tests tests/core_tests.cpp)
  target_link_libraries(rse-core-tests PRIVATE rse_core)
  add_test(NAME core-tests COMMAND rse-core-tests)
  add_executable(rse-nes-memory-tests tests/nes_memory_tests.cpp)
  target_link_libraries(rse-nes-memory-tests PRIVATE rse_nes)
  add_test(NAME nes-memory-tests COMMAND rse-nes-memory-tests)
endif()
if(RSE_BUILD_DESKTOP)
  find_package(SDL3 CONFIG REQUIRED)
  find_package(Vulkan REQUIRED)
  add_executable(rse-desktop apps/desktop/main.cpp)
  target_link_libraries(rse-desktop PRIVATE rse_core SDL3::SDL3 Vulkan::Vulkan)
endif()
