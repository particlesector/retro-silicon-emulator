cmake_minimum_required(VERSION 3.25)
project(RetroSiliconEmulator VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RSE_BUILD_TESTS "Build unit tests" ON)
option(RSE_BUILD_DESKTOP "Build SDL3 desktop frontend" OFF)
add_library(rse_core STATIC src/core/machine.cpp)
target_include_directories(rse_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(rse_core PUBLIC cxx_std_23)
add_library(rse_cpu6502 STATIC src/cpu/mos6502.cpp)
target_include_directories(rse_cpu6502 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(rse_cpu6502 PUBLIC cxx_std_23)
add_library(rse_ppu2c02 STATIC src/video/rp2c02.cpp)
target_include_directories(rse_ppu2c02 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(rse_ppu2c02 PUBLIC cxx_std_23)
add_library(rse_devices INTERFACE)
target_include_directories(rse_devices INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(rse_devices INTERFACE cxx_std_23)
add_library(rse_apu2a03 STATIC src/audio/rp2a03_apu.cpp)
target_include_directories(rse_apu2a03 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(rse_apu2a03 PUBLIC cxx_std_23)
add_library(rse_nes STATIC src/systems/nes/cartridge.cpp src/systems/nes/bus.cpp src/systems/nes/machine.cpp)
target_link_libraries(rse_nes PUBLIC rse_core rse_cpu6502 rse_ppu2c02 rse_apu2a03 rse_devices)
add_library(rse_cli_commands STATIC apps/cli/commands.cpp)
target_include_directories(rse_cli_commands PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/apps/cli)
target_link_libraries(rse_cli_commands PUBLIC rse_nes)
add_executable(rse-cli apps/cli/main.cpp)
target_link_libraries(rse-cli PRIVATE rse_cli_commands)
if(RSE_BUILD_TESTS)

  add_executable(rse-nes-audio-tests tests/nes_audio_tests.cpp)

    2a03 "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/nes6502/vectors.txt")

  target_link_libraries(rse-nes-system-tests PRIVATE rse_nes)

if(RSE_BUILD_DESKTOP)
  find_package(SDL3 CONFIG REQUIRED)
  find_package(Vulkan REQUIRED)
  add_executable(rse-desktop apps/desktop/main.cpp apps/desktop/audio_player.cpp)
  target_link_libraries(rse-desktop PRIVATE rse_nes SDL3::SDL3 Vulkan::Vulkan)
endif()
